Function RenumberArrayBracket(xmlString As String, ArrayNumber As Long) As String
    Dim regex As Object
    Dim newXmlString As String
    On Error GoTo ErrorHandler
    
    ' Create a regular expression object
    Set regex = CreateObject("VBScript.RegExp")
    regex.Global = True  ' Set this property to True to replace all matches
 
        regex.pattern = "\[\d+\]"
        newXmlString = regex.Replace(xmlString, "[" & ArrayNumber & "]")

    RenumberArrayBracket = newXmlString
Exit Function
ErrorHandler:
    Call LogError("RenumberArray", Err.Number, Err.Description)
    Resume Next
End Function

'------------------------------------------------------------------------------------
' SafeLogic
' Processes safety logic for PLC programming based on tag types and area names.
'
' Parameters:
'   rungString - The input string containing PLC rung logic
'   CopyNum - Number of copies to process
'   areaName - The area name to match in the dictionary
'
' Returns:
'   Modified rung string with safety logic applied
'------------------------------------------------------------------------------------
Function SafeLogic(rungString As String, CopyNum As Long, areaName As String) As String
    ' Dictionary sheet reference
    Dim sourceSheet As Worksheet
    Set sourceSheet = ThisWorkbook.Sheets("Dict")
    
    ' Variable declarations with clear naming
    Dim lastRow As Long, i As Long, tagIndex As Long
    Dim tagArray() As Variant, splitResult As Variant
    Dim splitText As String
    Dim ResultString As String, prefixString As String, tagType As String
    Dim rowsWithData As Object, rowNum As Variant
    
    ' Error handling
    On Error GoTo ErrorHandler
    
    ' Find the last row with data
    lastRow = lastRow ' no-op to preserve variable; lookups use cached data
    ' Determine the tag type and split text based on rungString content
    ' This section identifies what type of safety logic to apply
    If InStr(1, rungString, "ResetReady", vbTextCompare) > 0 Then
    tagArray = Array(21, 23)
    tagType = "O1"
    splitText = "XIC(Instruction.ChannelA) XIC(Instruction.ChannelB) XIO(Instruction.O1)"
    ElseIf InStr(1, rungString, ".FP", vbTextCompare) > 0 Then
        ' Fault Present logic
        tagArray = Array(21, 23, 27)
        tagType = "FP"
        splitText = "XIC(Instruction.FP)"
    ElseIf InStr(1, rungString, "NoEStop", vbTextCompare) > 0 Then
        ' Emergency Stop logic
        tagArray = Array(21)
        tagType = "O1"
        splitText = "XIC(Instruction.O1)"
    ElseIf InStr(1, rungString, "NoEPC", vbTextCompare) > 0 Then
        ' Emergency Pull Cord logic
        tagArray = Array(23)
        tagType = "O1"
        splitText = "XIC(Instruction.O1)"
    Else
        ' Default safety logic
        tagArray = Array(21, 23)
        tagType = "O1"
        splitText = "XIO(Instruction.O1) OTE(Safety_Area_Zone_Slncd_Trpd.0) ,"
    End If
    
    ' Split the string at the specified text
    splitResult = SplitStringAtInstruction(rungString, splitText)
    
    ' Separate the prefix and remaining rungString
    prefixString = splitResult(0)
    rungString = splitResult(1)
    
    ' Remove the original instruction from the rungString
    rungString = Replace(rungString, splitText, "")
    
    ' Get rows that match the area name
    Set rowsWithData = FindRowsWithMatch(ThisWorkbook.Sheets("Dict"), 5, areaName)
    
    ' Process tags and build the resultString
    tagIndex = 0
    ResultString = ""
    
    For i = LBound(tagArray) To UBound(tagArray)
        For Each rowNum In rowsWithData.Keys
            ' Only process rows that match the area name exactly
            If StrComp(areaName, DataCache.DictValue(rowNum, 5), vbTextCompare) = 0 Then
                ' Skip entries with "NO_" prefix
                If InStr(1, DataCache.DictValue(rowNum, tagArray(i)), "NO_", vbTextCompare) = 0 Then
                    ' Apply different logic based on tag type
                    Select Case tagType
                        Case "FP"
                            ' Fault Present logic
                            If tagArray(i) = 27 Then
                                ResultString = ResultString & "XIC(" & DataCache.DictValue(rowNum, tagArray(i)) & "_STO_CROUT" & "." & tagType & "),"
                            Else
                                ResultString = ResultString & "XIC(" & DataCache.DictValue(rowNum, tagArray(i)) & ".DCS" & "." & tagType & "),"
                            End If
                        Case "O1"
                            ' Output logic
                            If InStr(1, rungString, "ResetReady", vbTextCompare) <> 0 Then
                                ResultString = ResultString & "XIC(" & DataCache.DictValue(rowNum, tagArray(i)) & ".DCS.ChannelA)" & " XIC(" & DataCache.DictValue(rowNum, tagArray(i)) & ".DCS.ChannelB)" & " XIC(" & DataCache.DictValue(rowNum, tagArray(i)) & ".DCS.O1),"
                            ElseIf InStr(1, rungString, "NO", vbTextCompare) <> 0 Then
                                ResultString = ResultString & "XIC(" & DataCache.DictValue(rowNum, tagArray(i)) & ".DCS" & "." & tagType & ")"
                            Else
                                ResultString = ResultString & "XIO(" & DataCache.DictValue(rowNum, tagArray(i)) & ".DCS" & "." & tagType & ")OTE(Safety_Area_Zone_Slncd_Trpd." & tagIndex & "),"
                            End If
                    End Select
                    tagIndex = tagIndex + 1
                End If
            End If
        Next rowNum
    Next i
    
    ' Combine the parts based on tag type
    If tagType = "FP" Then
        ' For Fault Present, wrap the result in brackets
        ResultString = prefixString & "[" & ResultString & "]" & rungString
    Else
        ' For other types, concatenate directly
        ResultString = prefixString & ResultString & rungString
        ' Clean up any empty safety zone references
        ResultString = Replace(ResultString, ",OTE(Safety_Zone_Zone_Slncd_Trpd.0),", "", , , vbTextCompare)
    End If
    
    ' Replace area references
    ResultString = Replace(ResultString, DataCache.DictValue(17, 6), areaName)
    
    ' Return the final result
    SafeLogic = ResultString
    Exit Function

ErrorHandler:
    ' Log the error and continue
    Call LogError("SafeLogic", Err.Number, Err.Description)
    Resume Next
End Function

'------------------------------------------------------------------------------------
' SplitStringAtInstruction
' Splits a string at the specified instruction text and returns an array of two parts.
'
' Parameters:
'   xmlString - The input string to split
'   splitText - The text to split at
'
' Returns:
'   Array with two elements: [0] = text before split point, [1] = text from split point onward
'------------------------------------------------------------------------------------
Function SplitStringAtInstruction(xmlString As String, splitText As String) As Variant
    Dim parts(1) As String
    Dim splitIndex As Long
    
    On Error GoTo ErrorHandler
    
    ' Find the index of the split text
    splitIndex = InStr(xmlString, splitText)
    
    ' Check if the split text is found
    If splitIndex > 0 Then
        ' Split the string at the split text
        parts(0) = Left(xmlString, splitIndex - 1)
        parts(1) = Mid(xmlString, splitIndex)
    Else
        ' If the split text is not found, assign the whole string to parts(0) and empty string to parts(1)
        parts(0) = xmlString
        parts(1) = ""
    End If
    
    SplitStringAtInstruction = parts
    Exit Function
    
ErrorHandler:
    Call LogError("SplitStringAtInstruction", Err.Number, Err.Description)
    Resume Next
End Function

'------------------------------------------------------------------------------------
' ConvPI
' Processes conveyor-related Program Interface (PI) logic for PLC programming.
' Replaces conveyor placeholders with actual conveyor names from the dictionary.
'
' Parameters:
'   rungString - The input string containing PLC rung logic
'   areaName - The area name to match in the dictionary
'   CopyNum - Number of copies to process
'
' Returns:
'   Modified rung string with conveyor PI logic applied
'------------------------------------------------------------------------------------
Function ConvPI(rungString As String, areaName As String, CopyNum As Long) As String
    ' Dictionary sheet reference
    Dim sourceSheet As Worksheet
    Set sourceSheet = ThisWorkbook.Sheets("Dict")
    
    ' Variable declarations with clear naming
    Dim convNumber As Long, lastRow As Long
    Dim ProcessString As String
    Dim ResultString As String
    Dim rowsWithData As Object, rowNum As Variant
    
    ' Error handling
    On Error GoTo ErrorHandler
    
    ' Initialize variables
    lastRow = lastRow
    ProcessString = rungString
    convNumber = 20  ' Start with 20 conveyor inputs
    
    ' Get rows that match the area name in column 3
    Set rowsWithData = FindRowsWithMatch(ThisWorkbook.Sheets("Dict"), 3, areaName)

    ' Process each matching row
    For Each rowNum In rowsWithData.Keys
        ' If we've used all conveyor numbers and need more than one copy,
        ' append the current process string and reset for the next copy
        If CopyNum > 1 And convNumber = 0 Then
            ResultString = ProcessString & vbCrLf
            ProcessString = rungString
            ProcessString = Replace(ProcessString, "_Conv_PI1", "_Conv_PI2")
            convNumber = 20
        End If
        
        ' Only process rows that match the area name exactly
        If StrComp(areaName, DataCache.DictValue(rowNum, 3), vbTextCompare) = 0 Then
            ' Replace conveyor placeholder with actual conveyor name
            ProcessString = Replace(ProcessString, "Conv" & convNumber, DataCache.DictValue(rowNum, 1))
            convNumber = convNumber - 1
        End If
        
        ' Replace area references
        ProcessString = Replace(ProcessString, DataCache.DictValue(rowNum, 6), DataCache.DictValue(rowNum, 5))
        ProcessString = Replace(ProcessString, DataCache.DictValue(rowNum, 4), DataCache.DictValue(rowNum, 3))
    Next rowNum
    
    ' Determine the final result string based on copy number
    If CopyNum = 1 Then
        ResultString = ProcessString
    Else
        ResultString = ResultString & ProcessString
    End If
    
    ' Replace any remaining conveyor placeholders with NO_Conv
    For convNumber = 20 To 1 Step -1
        ResultString = Replace(ResultString, "Conv" & convNumber, "NO_Conv", , , vbTextCompare)
    Next convNumber

    ' Return the final result
    ConvPI = ResultString
    Exit Function

ErrorHandler:
    Call LogError("ConvPI", Err.Number, Err.Description)
    Resume Next
End Function
'------------------------------------------------------------------------------------
' SafePI
' Processes safety-related Program Interface (PI) logic for PLC programming.
' Handles emergency stop (ES) inputs and their left/right variants.
'
' Parameters:
'   rungString - The input string containing PLC rung logic
'   areaName - The area name to match in the dictionary
'   CopyNum - Number of copies to process
'
' Returns:
'   Modified rung string with safety PI logic applied
'------------------------------------------------------------------------------------
Function SafePI(rungString As String, areaName As String, CopyNum As Long) As String
    ' Dictionary sheet reference
    Dim sourceSheet As Worksheet
    Set sourceSheet = ThisWorkbook.Sheets("Dict")
    
    ' Variable declarations with clear naming
    Dim esNumber As Long, lastRow As Long
    Dim ProcessString As String
    Dim ResultString As String
    Dim cellVal As String
    Dim currentRow As Long
    Dim lookupRange As Range
    Dim foundCell As Range
    
    ' Error handling
    On Error GoTo ErrorHandler
    
    ' Initialize variables
    lastRow = sourceSheet.Range("G5").value
    ProcessString = rungString
    esNumber = 20  ' Start with 20 emergency stop inputs
    
    ' Process each row in the dictionary
    For currentRow = 17 To lastRow
        ' If we've used all ES numbers and need more than one copy,
        ' append the current process string and reset for the next copy
        If CopyNum > 1 And esNumber = 0 Then
            ResultString = ProcessString & vbCrLf
            ProcessString = rungString
            ProcessString = Replace(ProcessString, "_ES_PI", "_ES_PI2")
            esNumber = 20
        End If
        
        ' Only process rows that match the area name
        If StrComp(areaName, ThisWorkbook.Sheets("Dict").Cells(currentRow, 5).value, vbTextCompare) = 0 Then
            ' Process column 21 (Emergency Stop Button)
            If InStr(1, sourceSheet.Cells(currentRow, 21).value, "NO", vbTextCompare) = 0 Then
                ' Handle left/right variants
                If Right(sourceSheet.Cells(currentRow, 21).value, 1) = "L" Then
                    ' Create right variant from left
                    cellVal = CStr(ThisWorkbook.Sheets("Dict").Cells(currentRow, 21).value)
                    cellVal = Left(cellVal, Len(cellVal) - 1) & "R"
                    ProcessString = Replace(ProcessString, "Input" & esNumber, cellVal)
                    esNumber = esNumber - 1
                ElseIf Right(sourceSheet.Cells(currentRow, 21).value, 1) = "R" Then
                    ' Create left variant from right
                    cellVal = CStr(ThisWorkbook.Sheets("Dict").Cells(currentRow, 21).value)
                    cellVal = Left(cellVal, Len(cellVal) - 1) & "L"
                    ProcessString = Replace(ProcessString, "Input" & esNumber, cellVal)
                    esNumber = esNumber - 1
                End If
                
                ' Replace the input placeholder with the actual value
                ProcessString = Replace(ProcessString, "Input" & esNumber, sourceSheet.Cells(currentRow, 21).value, , , vbTextCompare)
                esNumber = esNumber - 1
            End If
            
            ' Process column 23 (Emergency Pull Cord)
            If InStr(1, sourceSheet.Cells(currentRow, 23).value, "NO", vbTextCompare) = 0 Then
                ' Handle left/right variants
                If Right(sourceSheet.Cells(currentRow, 23).value, 1) = "R" Then
                    ' Create left variant from right
                    cellVal = CStr(ThisWorkbook.Sheets("Dict").Cells(currentRow, 23).value)
                    cellVal = Left(cellVal, Len(cellVal) - 1) & "L"
                    ProcessString = Replace(ProcessString, "Input" & esNumber, cellVal)
                    esNumber = esNumber - 1
                ElseIf Right(sourceSheet.Cells(currentRow, 23).value, 1) = "L" Then
                    ' Create right variant from left
                    cellVal = CStr(ThisWorkbook.Sheets("Dict").Cells(currentRow, 23).value)
                    cellVal = Left(cellVal, Len(cellVal) - 1) & "R"
                    ProcessString = Replace(ProcessString, "Input" & esNumber, cellVal)
                    esNumber = esNumber - 1
                End If
                
                ' Replace the input placeholder with the actual value
                ProcessString = Replace(ProcessString, "Input" & esNumber, sourceSheet.Cells(currentRow, 23).value, , , vbTextCompare)
                esNumber = esNumber - 1
            End If
        End If
    Next currentRow
    
    ' Determine the final result string based on copy number
    If CopyNum = 1 Then
        ResultString = ProcessString
    Else
        ResultString = ResultString & ProcessString
    End If
    
    ' Replace any remaining input placeholders with NO_ESLS
    For esNumber = 20 To 1 Step -1
        ResultString = Replace(ResultString, "Input" & esNumber, "NO_ESLS")
    Next esNumber

    ' Replace area references
    ResultString = Replace(ResultString, "Main_Area_Safe", areaName)
    
    ' Look up the corresponding area name in column D
    Set lookupRange = ThisWorkbook.Sheets("Dict").Range("E17:E1007")
    Set foundCell = lookupRange.Find(What:=areaName, LookIn:=xlValues, LookAt:=xlWhole, MatchCase:=False, SearchFormat:=False)
    
    ' Replace Main_Area with the found value or handle not found case
    If Not foundCell Is Nothing Then
        ResultString = Replace(ResultString, "Main_Area", foundCell.Offset(0, -2).value)
    Else
        ' Log error instead of showing message box and ending execution
        Call LogError("SafePI", 0, "Area name '" & areaName & "' not found in dictionary")
        ' Use a default value instead of ending execution
        ResultString = Replace(ResultString, "Main_Area", "Unknown_Area")
    End If
    
    ' Return the final result
    SafePI = ResultString
    Exit Function

ErrorHandler:
    Call LogError("SafePI", Err.Number, Err.Description)
    Resume Next
End Function
Function FindCellInRowWithMatch(rowNum As Long, targetString As String) As Long
    Dim cell As Range
    Dim lastColumn As Long
    Dim ws As Worksheet
    On Error GoTo ErrorHandler
    ' If no worksheet is specified, use the active worksheet
    Set ws = ThisWorkbook.Sheets("Dict")
    lastColumn = 0
    ' Get the last column with data in the specified row
    lastColumn = ws.Cells(rowNum, ws.Columns.count).End(xlToLeft).Column
    
    ' Loop through each cell in the row
    For Each cell In ws.Range(ws.Cells(rowNum, 1), ws.Cells(rowNum, lastColumn))

    Next cell
    
    ' Return "Not Found" if no match is found
    FindCellInRowWithMatch = 0
Exit Function

' retriveTypes
' Determines which type of tag is present in a rung by searching the relevant sheets.
'------------------------------------------------------------------------------------
Sub retriveTypes(rungString As String, RoutineValue As String, ByRef tagType As String, ByRef colNum As Object)
    Dim tagArray() As Variant
    Dim sourceSheet As Worksheet

    On Error GoTo ErrorHandler

    If InStr(1, RoutineValue, "CP", vbTextCompare) > 0 Then
        tagArray = Array(1)
        Set sourceSheet = ThisWorkbook.Sheets("IO Dict")
    Else
        tagArray = Array(10, 12, 14, 2, 16, 18, 20, 22, 24, 26, 28, 30, 32, 34, 4, 6)
        Set sourceSheet = ThisWorkbook.Sheets("Dict")
    End If
    
    'If InStr(1, rungString, "MS_Run_Safe", vbTextCompare) > 0 Then
    '   MsgBox "Input String" & rungString, vbExclamation, "MS Diag"
    'End If

    Dim Results As String
    Dim RetString As String
    Dim i As Integer

    For i = LBound(tagArray) To UBound(tagArray)
        If Results = "" Then
            Results = FindTagTypeInText(rungString, sourceSheet, tagArray(i))
            
            If colNum Is Nothing Then
                Set colNum = CreateObject("Scripting.Dictionary")
            End If
            
            If colNum.count = 0 And Results <> "" Then
                tagType = Results
                colNum.Add tagArray(i), tagArray(i)
            End If
        Else
            RetString = FindTagTypeInText(rungString, sourceSheet, tagArray(i))
            If RetString = Results Then
                If Not colNum.Exists(tagArray(i)) Then
                    colNum.Add tagArray(i), tagArray(i)
                End If
            End If
        End If
    Next i
    Exit Sub

ErrorHandler:
    Call LogError("retriveTypes", Err.Number, Err.Description)
    Resume Next
End Sub

'------------------------------------------------------------------------------------
' RenumberLines
' Rewrites "Number=""X""" attributes from 0..N in ascending order.
'------------------------------------------------------------------------------------
Function RenumberLines(ByVal xmlString As String) As String
    On Error GoTo ErrHandler
    
    Dim regex As Object
    Set regex = CreateObject("VBScript.RegExp")
    regex.pattern = "Number=""\d+"""
    regex.Global = True
    
    Dim lines As Variant
    lines = Split(xmlString, vbCrLf)
    
    Dim i As Long, newNumber As Long
    newNumber = 0
    
    For i = LBound(lines) To UBound(lines)
        If regex.Test(lines(i)) Then
            lines(i) = regex.Replace(lines(i), "Number=""" & newNumber & """")
            newNumber = newNumber + 1
        End If
    Next i
    
    RenumberLines = Join(lines, vbCrLf)
    Exit Function
    
ErrHandler:
    LogError "RenumberLines", Err.Number, Err.Description
    Resume Next
End Function

Function ModuleCommDiag(rungString As String, CopyNum As Long) As String
    Dim ResultString As String
    Dim ProcessString As String
    Dim tempType As String
    Dim dict As Object
    Dim AddDict As Object
    Dim key  As Variant
    Dim i As Long
    On Error GoTo ErrorHandler
    ResultString = ""
    
    If InStr(1, rungString, "NOP", vbTextCompare) > 0 And CopyNum > 1 Then
        ResultString = rungString & vbCrLf
        If InStr(1, rungString, "Group", vbTextCompare) > 0 Then
            For i = 2 To CopyNum
                ProcessString = rungString
                ProcessString = Replace(ProcessString, "Group1", "Group" & i)
                ResultString = ResultString & ProcessString & vbCrLf
            Next i
        End If
        
    ElseIf InStr(1, rungString, "AOI", vbTextCompare) > 0 Then
        Set dict = CreateObject("Scripting.Dictionary")
        Set AddDict = CreateObject("Scripting.Dictionary")
        tempType = templateCheckCaseSensitiveWholeWord(rungString)
        'MsgBox "type: " & tempType & vbCrLf & "string: " & RungString
        ' Pass Temptype to the dictionary builder
        BuildModuleDictionaryUsingSearch dict, tempType
        
        For Each key In dict.Keys
            ProcessString = rungString
            If InStr(1, ProcessString, dict(key)(0), vbTextCompare) > 0 And _
               InStr(1, ResultString, key, vbTextCompare) = 0 And _
               Not AddDict.Exists(ProcessString) Then
                   
                ProcessString = Replace(ProcessString, dict(key)(0), key)
                ProcessString = Replace(ProcessString, "Main_Area", dict(key)(2))
                ProcessString = Replace(ProcessString, "UpStrmComm", dict(key)(1))
                ResultString = ResultString & ProcessString & vbCrLf
                AddDict.Add ProcessString, ""
            End If
        Next key
    End If
    
    ModuleCommDiag = ResultString
    Exit Function
ErrorHandler:
    Call LogError("ModuleCommDiag", Err.Number, Err.Description)
    Resume Next
End Function
Function templateCheckCaseSensitiveWholeWord(RungStringCopy As String) As String
    Dim rng As Range
    Dim cell As Range
    Dim re As Object
    
    Set rng = ThisWorkbook.Names("IOConfigRSLT").RefersToRange
    
    ' Create the RegExp object via late binding
    Set re = CreateObject("VBScript.RegExp")
    re.ignoreCase = False     ' Case-sensitive
    re.Global = False         ' Only need the first match
    
    For Each cell In rng
        ' We define "word chars" as letters, digits, or underscore.
        ' So we want to ensure a match is bounded by the start/end of string
        ' or a non-word char ([^A-Za-z0-9_]) on both sides:
        re.pattern = "(^|[^A-Za-z0-9_])" & cell.value & "([^A-Za-z0-9_]|$)"
        
        If re.Test(RungStringCopy) Then
            templateCheckCaseSensitiveWholeWord = cell.value
            Exit Function
        End If
    Next cell
    
    ' If none matched, return empty
    templateCheckCaseSensitiveWholeWord = ""
End Function

Sub BuildModuleDictionaryUsingSearch(ByRef dict As Object, ByVal tempType As String)
    Dim wsIO As Worksheet
    Dim wsDict As Worksheet
    Dim wsConfig As Worksheet
    Dim lookupRange As Range
    Dim ioLastRow As Long, dictLastRow As Long, i As Long, j As Long, k As Long
    Dim moduleName As String
    Dim moduleType As String
    Dim moduleTemp As Variant
    Dim parent As String
    Dim moduleArea As Variant
    Dim found As Boolean

    ' Initialize the dictionary if needed
    If dict Is Nothing Then
        Set dict = CreateObject("Scripting.Dictionary")
    End If

    Set wsIO = ThisWorkbook.Sheets("IO")
    Set wsDict = ThisWorkbook.Sheets("Dict")
    Set wsConfig = ThisWorkbook.Sheets("Config")
    Set lookupRange = wsConfig.Range("C34:D80")
    
    ioLastRow = wsIO.Cells(wsIO.Rows.count, "B").End(xlUp).Row
    dictLastRow = wsDict.Range("G5").value

    For i = 6 To ioLastRow
        moduleName = Trim(wsIO.Cells(i, 2).value)
        moduleType = wsIO.Cells(i, 3).value
        parent = wsIO.Cells(i, 6).value
        
        moduleTemp = Application.VLookup(moduleType, lookupRange, 2, False)
        
        If parent = "" Then
            parent = "NO_CommDev"
        End If

        moduleArea = "Area_Unkown"
        found = False
        For j = 17 To dictLastRow
            For k = 41 To 60
                If InStr(1, Trim(wsDict.Cells(j, k).value), moduleName, vbTextCompare) > 0 Then
                    moduleArea = wsDict.Cells(j, 3).value
                    found = True
                    Exit For
                End If
            Next k
            If found Then Exit For
        Next j

        ' Only add this module if its template matches the given tempType.
        If moduleTemp = tempType Then
            If Not dict.Exists(moduleName) And InStr(1, moduleName, "PLC", vbTextCompare) = 0 Then
                dict.Add moduleName, Array(moduleTemp, parent, moduleArea)
            End If
        End If
    Next i
End Sub
